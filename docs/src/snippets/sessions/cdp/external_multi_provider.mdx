from typing import Literal

from kernel import Kernel
from notte_sdk import NotteClient

client = NotteClient()


def run_automation(provider: Literal["notte", "kernel"], url: str, instructions: str):
    """Run automation on different browser providers"""

    if provider == "notte":
        # Use Notte's browser
        with client.Session() as session:
            session.execute(type="goto", url=url)
            return session.scrape(instructions=instructions)

    elif provider == "kernel":
        # Use Kernel's browser
        kernel = Kernel()
        kernel_browser = kernel.browsers.create()

        try:
            with client.Session(cdp_url=kernel_browser.cdp_ws_url) as session:
                session.execute(type="goto", url=url)
                return session.scrape(instructions=instructions)
        finally:
            kernel.browsers.delete_by_id(kernel_browser.session_id)


# Run on Notte's infrastructure
result1 = run_automation("notte", "https://example.com", "Get the latest news")
print(f"Notte result: {result1}")

# Run on Kernel's infrastructure
result2 = run_automation("kernel", "https://example.com", "Get the latest news")
