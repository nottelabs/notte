from notte_sdk import NotteClient


def run():
    """Check if all schedules are healthy."""
    client = NotteClient()

    # Get all functions and check their recent runs
    functions = client.functions.list()

    health_report = []
    for function in functions.items:
        runs = client.functions.list_runs(function.function_id)

        # Check recent runs
        recent_failures = [r for r in runs.items if r.status == "failed"]

        health_report.append({"function_id": function.function_id, "recent_failures": len(recent_failures)})

    return health_report
